\begin{abstract}\noindent
One change to the maintenance model --- initialising the Ornstein--Uhlenbeck
maintenance noise at its stationary law rather than at zero --- makes the transient
negligible over the delay-period analysis window under three conditions that can be
argued on their own terms. The rate is then the \emph{stationary} Kac--Rice rate, with
an explicit error bar, and every prediction follows in a line.
Section~\ref{sec:model} states the model and the rate, Section~\ref{sec:approx} the
approximation theorem and its conditions, Section~\ref{sec:proof} the proof,
Section~\ref{sec:pred} the predictions. All statements are verified numerically
(Section~\ref{sec:checks}).
\end{abstract}

\section{The model and the rate}\label{sec:model}

Everything before stimulus offset is unchanged. Maintenance begins at
$\tau=t_{\mathrm{offset}}+L_{\mathrm{motion}}$, at which time the gate-facing signal
$y$ equals the maintained target $m$. During maintenance,
\begin{equation}
\dd\varepsilon(t)=-\kappa\,\varepsilon(t)\,\dd t+\sigma_\varepsilon\,\dd W_\varepsilon(t),
\qquad
\dd y(t)=\lambda\,[\,m+\varepsilon(t)-y(t)\,]\,\dd t ,
\label{eq:sde}
\end{equation}
and $H:=y-m$ obeys $\dd H=\lambda(\varepsilon-H)\dd t$.

\paragraph{The change.}
$H(\tau)=0$ is \emph{forced}: the filter output is continuous and $m$ is defined as
the value of $y$ at $\tau$. $\varepsilon(\tau)$ is free; the present model sets it to
$0$. We propose instead
\begin{equation}
\varepsilon(\tau)\sim\mathrm{N}\!\left(0,\ \sigma_\varepsilon^2/2\kappa\right)
\quad\text{independent of }W_\varepsilon,
\tag{\textsc{ss}}\label{eq:warm}
\end{equation}
the stationary law of the first equation in \eqref{eq:sde}.

\paragraph{Why.}
$\varepsilon$ represents fluctuations intrinsic to holding a value; nothing makes such
a process sit exactly at its mean at the instant maintenance begins, and no
measurement constrains it. The choice matters because it is the \emph{only} source of
a slow transient in the model. The covariance of $(\varepsilon,H)$ relaxes as a
combination of three modes, $e^{-2\kappa\rho}$, $e^{-(\lambda+\kappa)\rho}$ and
$e^{-2\lambda\rho}$ (Remark~\ref{rem:modes}); the cold start excites all three and the
first can be as slow as $1/\kappa\sim5$~s, so the analysis window lies inside the
transient and every prediction is asymptotic or conditional. Under \eqref{eq:warm} the
first mode has zero coefficient and relaxation is at least as fast as $\lambda$.

\paragraph{Notation.}
$\rho:=t-\tau$; $\beta/\delta$ is the speed of temporal preparation, with $\beta=0$
under Model~1; $d(t)=g-|m|-(\beta/\delta)(t-t_{\mathrm{offset}})^{+}$ is the distance
to the gate; $\varphi,\Phi$ are the standard normal density and c.d.f., and
$g(u):=u\Phi(u)+\varphi(u)$. Since $\varphi'(u)=-u\varphi(u)$, the two middle terms of
$g'(u)=\Phi(u)+u\varphi(u)+\varphi'(u)$ cancel, so
\begin{equation}
g'=\Phi,\qquad g''=\varphi ,
\label{eq:gprime}
\end{equation}
and $g$ is positive, strictly increasing and strictly convex, with
$g(0)=1/\sqrt{2\pi}$. Write
$s^{2}:=\sigma_\varepsilon^{2}/2\kappa$ and
\begin{equation}
\Ain:=\frac{\lambda s^{2}}{\lambda+\kappa},\qquad
\Bin:=\frac{\lambda^{2}\sigma_\varepsilon^{2}}{2(\lambda+\kappa)},\qquad
\omega:=\sqrt{\Bin/\Ain}=\sqrt{\lambda\kappa},\qquad
a_\infty:=\frac{d}{\sqrt{\Ain}} .
\label{eq:stat}
\end{equation}
These are the stationary variance of $H$, the stationary variance of $\dot H$, the
characteristic frequency, and the standardised distance to the gate; at stationarity
$\Cov(H,\dot H)=0$.

\paragraph{The rate.}
Let $Z:=s_{\mathrm{resp}}H-d$, so partial activations are upcrossings of zero by $Z$
and $\dot Z=\dot H+\beta/\delta$. The Kac--Rice formula for a differentiable Gaussian
process --- which requires no stationarity --- gives
\begin{equation}
\nu(t)=p_{Z(t)}(0)\,\E\bigl[\dot Z\mathbf{1}_{\{\dot Z>0\}}\mid Z=0\bigr]
=\frac{\varphi(a)}{\sigma_H}\,G,
\qquad G:=\E\bigl[\dot Z\mathbf{1}_{\{\dot Z>0\}}\mid Z=0\bigr],
\label{eq:nu}
\end{equation}
with $a:=d/\sigma_H$. Since $(Z,\dot Z)$ is jointly Gaussian, $\dot Z$ given $Z=0$ is
$\mathrm{N}(M,S)$ with
\begin{equation}
M=\frac{\beta}{\delta}+\frac{c_H}{\sigma_H^{2}}\,d,
\qquad
S=\sigma_{\dot H}^{2}-\frac{c_H^{2}}{\sigma_H^{2}},
\qquad
u:=\frac{M}{\sqrt S},
\label{eq:MSu}
\end{equation}
and the expected positive part of an $\mathrm{N}(M,S)$ variable is
$M\Phi(u)+\sqrt S\,\varphi(u)$. Substituting $u=M/\sqrt S$ into $g$ gives
$\sqrt S\,g(u)=\sqrt S\,(M/\sqrt S)\Phi(u)+\sqrt S\,\varphi(u)$, so
\begin{equation}
G=M\Phi(u)+\sqrt S\,\varphi(u)=\sqrt S\,g(u).
\label{eq:Gid}
\end{equation}
Writing $G$ this way is what makes the estimates of Section~\ref{sec:proof} work: it
separates the scale $\sqrt S$ from the shape $g(u)$, and $g$ has the two properties
used there --- $g'=\Phi\le1$ and $\log g$ concave.

\begin{proposition}[the stationary rate]\label{prop:statrate}
Write $\nustat$ for \eqref{eq:nu} evaluated at the stationary moments \eqref{eq:stat},
that is with $\sigma_H^{2}=\Ain$, $\sigma_{\dot H}^{2}=\Bin$, $c_H=0$. Then
\begin{equation}
\nustat=\varphi(a)\left[\frac{\beta}{\delta\sigma_H}\Phi\!\left(\frac{\beta}{\delta\sigma_{\dot H}}\right)
+\omega\,\varphi\!\left(\frac{\beta}{\delta\sigma_{\dot H}}\right)\right],
\label{eq:nuasymp}
\end{equation}
whose bracket does not depend on $d$. Consequently:
\begin{enumerate}
\item[(i)] \textbf{Model 2} ($\beta>0$): $a(\rho)=a(0)-\beta\rho/(\delta\sigma_H)$ is
affine, so exactly
\begin{equation}
\frac{\dd\ln\nustat}{\dd\rho}=\frac{\beta}{\delta\sigma_H}\,a(\rho)>0,
\qquad
\frac{\dd^{2}\ln\nustat}{\dd\rho^{2}}=-\left(\frac{\beta}{\delta\sigma_H}\right)^{\!2} .
\label{eq:slope}
\end{equation}
\item[(ii)] \textbf{Model 1} ($\beta=0$): $M=0$, $u=0$, $a$ is constant, and
\begin{equation}
\nustat=\frac{\omega}{2\pi}\,e^{-a^{2}/2},
\qquad
\frac{\dd\ln\nustat}{\dd\rho}\equiv0 .
\label{eq:m1flat}
\end{equation}
\item[(iii)] Under both models $\nustat=\varphi(a)\times(\text{a factor independent of
}d)$, so $\nustat$ is strictly decreasing in $a$, with no condition.
\end{enumerate}
\end{proposition}
\begin{proof}
Setting $c_H=0$ in \eqref{eq:nu} gives \eqref{eq:nuasymp} and
$\ln\nustat=\mathrm{const}-a^{2}/2$, whence \eqref{eq:slope} since $a$ is affine, and
(iii). For (ii), $\beta=0$ gives $M=0$, $g(0)=1/\sqrt{2\pi}$,
$G=\sqrt{\Bin}/\sqrt{2\pi}=\omega\sigma_H/\sqrt{2\pi}$, so
$\nustat=\varphi(a)\omega/\sqrt{2\pi}=\omega e^{-a^{2}/2}/2\pi$, constant in $\rho$.
\end{proof}

\section{The stationary approximation}\label{sec:approx}

\begin{theorem}\label{thm:main}
Assume
\begin{equation}
\text{\emph{(i)}}\ \lambda\ge5\,\mathrm{s}^{-1},\qquad
\text{\emph{(ii)}}\ \kappa\ge0.2\,\mathrm{s}^{-1},\qquad
\text{\emph{(iii)}}\ a_\infty\le5 .
\label{eq:conds}
\end{equation}
Then throughout the analysis window $\rho\in[1,3]$~s, and for both models:
\begin{enumerate}
\item[\emph{(a)}] $\nu(\rho)=\nustat(\rho)\,e^{\theta(\rho)}$ with
$|\theta(\rho)|\le\mathcal{E}(\rho)$, where $\mathcal{E}(1)\le0.36$ and
$\mathcal{E}$ decays like $e^{-\lambda\rho}$: the rate is exact to within a factor
$1.43$ at the start of the window and $1.0004$ at its end;
\item[\emph{(b)}] the least-squares slope $c_1$ and quadratic coefficient $c_2$ of
$\ln\nu$ on the $20$ bins of the window satisfy
\[
\bigl|c_1-c_1^{\mathrm{stat}}\bigr|\le0.081\ \mathrm{s}^{-1},
\qquad
\bigl|c_2-c_2^{\mathrm{stat}}\bigr|\le0.145\ \mathrm{s}^{-2};
\]
\item[\emph{(c)}] under $\lambda\ge6$ these become $0.027\,\mathrm{s}^{-1}$ and
$0.049\,\mathrm{s}^{-2}$; under $\lambda\ge8$, $0.0030$ and $0.0058$.
\end{enumerate}
\end{theorem}

\noindent The observed log-rate slope is $+0.867\,\mathrm{s}^{-1}$, so (b) is a $9\%$
error bar on it and (c) a $3\%$ one. \emph{From Section~\ref{sec:pred} on we assume
\eqref{eq:conds}, work with $\nustat$, and carry (b) as the error bar.}

\subsection{The three conditions}\label{sec:conds}

Each is argued without reference to the measured rate.

\begin{description}
\item[(i) $\lambda\ge5\,\mathrm{s}^{-1}$, i.e.\ $1/\lambda\le200$~ms.]
$1/\lambda$ is the time constant of the low-pass filter carrying motor preparation to
the gate. Dendauw et al.\ estimate $\lambda\approx20\,\mathrm{s}^{-1}$ ($50$~ms) in an
immediate-report task, so \eqref{eq:conds} is four times more permissive than the
published value. $\lambda$ is in principle identifiable from these data, since it sets
how fast the rate follows the stimulus; that check has not been carried out.
\item[(ii) $\kappa\ge0.2\,\mathrm{s}^{-1}$, i.e.\ $1/\kappa\le5$~s.]
$1/\kappa$ is the correlation time of the maintenance noise:
$\Cov(\varepsilon(t),\varepsilon(t+r))\propto e^{-\kappa|r|}$ at stationarity, so it is
the lag at which the fluctuation has decorrelated to $1/e$. Requiring it to be at most
one trial says that what is maintained fluctuates \emph{within} a trial; a longer
correlation time describes a drift carried across trials, for which the model has no
term and which the randomised design averages out rather than expresses as a
delay-period ramp. (i) and (ii) are substitutes: $\lambda\to\lambda+1$ buys about what
$\kappa\to2\kappa$ buys, so only one needs to be argued hard.
\item[(iii) $a_\infty\le5$.] Not fitted. By \eqref{eq:m1flat} the expected number of
partial activations per $2$~s delay window is $(\omega/\pi)e^{-a^{2}/2}$; at the
largest $\omega$ in play that is one per $19$ trials at $a=3.3$, one per $246$ at
$a=4$, and \emph{one per $22\,000$ at $a=5$}. So (iii) says only that the model must
produce partial activations at all --- the experiment comprises of order $10^{4}$
trials --- not that it must reproduce the observed rate. Inverting \eqref{eq:m1flat} at
the observed rates would give the sharper $a\le3.3$, tightening every bound by about a
third; it is not used.
\end{description}

\section{Proof of Theorem~\ref{thm:main}}\label{sec:proof}

The proof runs through four lemmas: the exact moments, a bound on how far they are
from stationary, the resulting pointwise error on $\ln\nu$, and its transfer to the
fitted coefficients. Nothing in this section is used elsewhere.

\begin{lemma}[exact moments]\label{lem:moments}
Under \eqref{eq:sde} and \eqref{eq:warm} with $H(\tau)=0$, put
$E_1:=e^{-(\lambda+\kappa)\rho}$, $E_2:=e^{-2\lambda\rho}$ and
\begin{equation}
X(\rho):=\frac{(\lambda+\kappa)(E_2-E_1)}{\lambda-\kappa}
\qquad\bigl(=-2\lambda\rho\,E_2\ \text{ when }\lambda=\kappa\bigr).
\label{eq:Xdef}
\end{equation}
Then for every $\rho>0$
\begin{equation}
\frac{\sigma_H^{2}}{\Ain}=1+X-E_1,\qquad
\frac{c_H}{\Ain}=-\lambda X,\qquad
\frac{\sigma_{\dot H}^{2}}{\Bin}=1+\frac{\lambda}{\kappa}\bigl(X+E_1\bigr).
\label{eq:moments}
\end{equation}
\end{lemma}
\begin{proof}
$(\varepsilon,H)$ is a linear diffusion with
$F=\bigl(\begin{smallmatrix}-\kappa&0\\ \lambda&-\lambda\end{smallmatrix}\bigr)$ and
$Q=\mathrm{diag}(\sigma_\varepsilon^{2},0)$, so its covariance solves
$\dot P=FP+PF^{\!\top}+Q$ with $P(0)=\mathrm{diag}(s^{2},0)$. The system is triangular:
\[
\dot P_{\varepsilon\varepsilon}=-2\kappa P_{\varepsilon\varepsilon}+\sigma_\varepsilon^{2},
\qquad
\dot P_{\varepsilon H}=-(\kappa+\lambda)P_{\varepsilon H}+\lambda P_{\varepsilon\varepsilon},
\qquad
\dot P_{HH}=2\lambda(P_{\varepsilon H}-P_{HH}),
\]
all starting from $(s^{2},0,0)$. Since
$\dot P_{\varepsilon\varepsilon}(0)=-2\kappa s^{2}+\sigma_\varepsilon^{2}=0$, the
stationary start freezes the first component: $P_{\varepsilon\varepsilon}\equiv s^{2}$.
Then $P_{\varepsilon H}=\Ain(1-E_1)$ and
\[
P_{HH}=2\lambda\!\int_0^{\rho}\!e^{-2\lambda(\rho-v)}P_{\varepsilon H}(v)\,\dd v
=\Ain\left[1-E_2-2\lambda\frac{E_1-E_2}{\lambda-\kappa}\right],
\]
and writing $-2\lambda=-(\lambda+\kappa)-(\lambda-\kappa)$ in the numerator gives
$P_{HH}/\Ain-1=X-E_1$. Since $\dot H=\lambda(\varepsilon-H)$,
$c_H=\lambda(P_{\varepsilon H}-P_{HH})=-\lambda\Ain X$, and
$\sigma_{\dot H}^{2}=\lambda^{2}(P_{\varepsilon\varepsilon}-2P_{\varepsilon H}+P_{HH})
=\Bin+\lambda^{2}\Ain(X+E_1)$ using $\lambda^{2}(s^{2}-\Ain)=\Bin$ and
$\lambda^{2}\Ain/\Bin=\lambda/\kappa$. The $\lambda=\kappa$ form follows by
l'H\^opital.
\end{proof}

\begin{remark}[the three modes]\label{rem:modes}
The operator $P\mapsto FP+PF^{\!\top}$ has eigenvalues $\{-2\kappa,-(\lambda+\kappa),-2\lambda\}$
--- the pairwise sums of the eigenvalues of $F$ --- so a general initial condition
relaxes as a combination of $e^{-2\kappa\rho}$ (the $\varepsilon$--$\varepsilon$
block), $e^{-(\lambda+\kappa)\rho}$ (the cross term) and $e^{-2\lambda\rho}$ (the
$H$--$H$ block). Under \eqref{eq:warm} the first has zero coefficient, because
$P_{\varepsilon\varepsilon}$ is constant, and \eqref{eq:moments} contains only $E_1$
and $E_2$. The slower of those two is $e^{-\mu\rho}$ with
$\mu:=\lambda+\min(\lambda,\kappa)\ge\lambda$; it equals $e^{-2\lambda\rho}$ only when
$\kappa\ge\lambda$. Relaxation is therefore at least as fast as $\lambda$, uniformly
in $\kappa$.
\end{remark}

\begin{lemma}[distance from stationarity]\label{lem:bounds}
With $\mu:=\lambda+\min(\lambda,\kappa)$ and
\begin{equation}
\mathcal{D}:=\min\left\{\rho\,e^{-\mu\rho},\ \frac{E_1+E_2}{|\lambda-\kappa|}\right\}
\quad(\text{first entry only when }\lambda=\kappa),
\label{eq:Dcal}
\end{equation}
put
\begin{equation}
\bar\Delta:=(\lambda+\kappa)\mathcal{D}+E_1,\quad
\bar\gamma:=\lambda(\lambda+\kappa)\mathcal{D},\quad
\bar e:=\min\left\{\frac{\lambda}{\kappa}\bar\Delta,\
\frac{\lambda\bigl[(\lambda+\kappa)E_2+2\kappa E_1\bigr]}{\kappa|\lambda-\kappa|}\right\}.
\label{eq:Dbar}
\end{equation}
Then
$\bigl|\sigma_H^{2}/\Ain-1\bigr|\le\bar\Delta$,
$0\le c_H/\Ain\le\bar\gamma$ and
$\bigl|\sigma_{\dot H}^{2}/\Bin-1\bigr|\le\bar e$, all decreasing on $\rho\ge1$ when
$\lambda\ge1$.
\end{lemma}
\begin{proof}
Everything reduces to bounding $|E_1-E_2|/|\lambda-\kappa|$, since by
Lemma~\ref{lem:moments} $X=(\lambda+\kappa)(E_2-E_1)/(\lambda-\kappa)$ and, adding and
subtracting $(\lambda+\kappa)E_1$ in the numerator,
$\sigma_{\dot H}^{2}/\Bin-1=(\lambda/\kappa)\bigl[(\lambda+\kappa)E_2-2\kappa E_1\bigr]/(\lambda-\kappa)$.
Two bounds are available and we take the smaller. \emph{(a)} By the mean value theorem
applied to $x\mapsto e^{-x\rho}$ on the interval with endpoints $2\lambda$ and
$\lambda+\kappa$ --- whose length is $|\lambda-\kappa|$ and whose left end is $\mu$ ---
$|E_1-E_2|/|\lambda-\kappa|\le\rho e^{-\mu\rho}$. \emph{(b)} Trivially
$|E_1-E_2|\le E_1+E_2$. Entry (a) is sharp when $\lambda\approx\kappa$, where (b) is
singular; (b) is sharp when the rates are far apart, where (a)'s worst-case exponent
$\mu$ is pessimistic. The sign $X\le0$ follows by inspection of \eqref{eq:Xdef}, giving
$c_H\ge0$. The second entry of $\bar e$ comes from bounding the two exponentials
separately. For monotonicity,
$\frac{\dd}{\dd\rho}(1+c\rho)e^{-\mu\rho}=e^{-\mu\rho}(c-\mu-\mu c\rho)$ with
$c=\lambda+\kappa\ge\mu$ is negative for $\rho>(c-\mu)/(\mu c)$, and
$(c-\mu)/(\mu c)\le1/(2\lambda)\le1/2$ when $\lambda\ge1$; the remaining entries are
sums of decaying exponentials.
\end{proof}

\begin{lemma}[pointwise error]\label{lem:point}
Fix $\rho\ge1$, evaluate $\bar\Delta,\bar\gamma,\bar e$ at $\rho$, and write
$u_s:=\beta/(\delta\sqrt{\Bin})$, $c_g:=\Phi(0)/g(0)=1.2533$ and
$\eta:=\bar e+\bar\gamma^{2}/\bigl(\omega^{2}(1-\bar\Delta)\bigr)$.
If $\bar\Delta<1$ and $\eta<1$ then, for both models,
\begin{equation}
\bigl|\ln\nu(\rho)-\ln\nustat(\rho)\bigr|\le\mathcal{E}(\rho):=
\frac{(a_\infty^{2}+1)\bar\Delta}{2(1-\bar\Delta)}
+\frac{\eta}{2(1-\eta)}
+c_g\!\left[\frac{\bar\gamma\,a_\infty}{\omega(1-\bar\Delta)\sqrt{1-\eta}}
+u_s\!\left(\frac{1}{\sqrt{1-\eta}}-1\right)\right].
\label{eq:point}
\end{equation}
\end{lemma}
\begin{proof}
Write $\Delta:=\sigma_H^{2}/\Ain-1$, $\gamma:=c_H/\Ain$,
$E:=\sigma_{\dot H}^{2}/\Bin-1$, bounded by Lemma~\ref{lem:bounds}. From
\eqref{eq:nu},
\[
\ln\nu-\ln\nustat=-\tfrac12\bigl(a^{2}-a_\infty^{2}\bigr)-\tfrac12\ln\frac{\sigma_H^{2}}{\Ain}
+\ln\frac{G}{G_s},\qquad G=\sqrt S\,g(u),\quad G_s=\sqrt{\Bin}\,g(u_s).
\]
Throughout, $|\ln(1+x)|\le|x|/(1-|x|)$ for $|x|<1$.

\emph{(1)} $a^{2}=a_\infty^{2}/(1+\Delta)$, so
$|a^{2}-a_\infty^{2}|\le a_\infty^{2}\bar\Delta/(1-\bar\Delta)$; halving and adding
$\tfrac12|\ln(1+\Delta)|\le\bar\Delta/(2(1-\bar\Delta))$ gives the first term.

\emph{(2)} $S=\Bin\bigl[(1+E)-\gamma^{2}/(\omega^{2}(1+\Delta))\bigr]$ since
$\Ain/\Bin=\omega^{-2}$, so $|S/\Bin-1|\le\eta$, hence $S\ge\Bin(1-\eta)>0$ and
$\tfrac12|\ln(S/\Bin)|\le\eta/(2(1-\eta))$: the second term.

\emph{(3)} $(\ln g)'=\Phi/g$ and $(\ln g)''=-D/g^{2}<0$, where
$D:=\Phi^{2}-\varphi g$ satisfies $D'=\varphi(\Phi+ug)>0$ and $D(-\infty)=0$, so
$D>0$; thus $\Phi/g$ is decreasing and $\sup_{u\ge0}(\ln g)'=\Phi(0)/g(0)=c_g$. Since
$M=\beta/\delta+\gamma d/(1+\Delta)\ge\beta/\delta\ge0$ we have $u,u_s\ge0$, so
$|\ln g(u)-\ln g(u_s)|\le c_g|u-u_s|$. Splitting
$u-u_s=\frac{M-\beta/\delta}{\sqrt S}+\frac{\beta}{\delta}\bigl(\frac{1}{\sqrt S}-\frac{1}{\sqrt{\Bin}}\bigr)$:
the first piece is at most
$\bar\gamma a_\infty/\bigl(\omega(1-\bar\Delta)\sqrt{1-\eta}\bigr)$, using
$M-\beta/\delta=\gamma d/(1+\Delta)$, $d=a_\infty\sqrt{\Ain}$ and
$\sqrt S\ge\sqrt{\Bin(1-\eta)}$; the second equals $u_s|(1+x)^{-1/2}-1|$ with
$|x|\le\eta$, hence at most $u_s(1/\sqrt{1-\eta}-1)$. Multiplying by $c_g$ gives the
third term.
\end{proof}

\begin{lemma}[transfer to the fitted coefficients]\label{lem:transfer}
On the $20$ bin centres $\rho_1,\dots,\rho_{20}$ of the window, the least-squares
slope and quadratic coefficient are linear functionals $c_j=\sum_i w^{(j)}_i f_i$ of
the profile, with $\sum_i|w^{(1)}_i|=1.504$ and $\sum_i|w^{(2)}_i|=2.916$. Hence
\begin{equation}
\bigl|c_j-c_j^{\mathrm{stat}}\bigr|\le\sum_i\bigl|w^{(j)}_i\bigr|\,\mathcal{E}(\rho_i)
\ \le\ \Bigl(\textstyle\sum_i\bigl|w^{(j)}_i\bigr|\Bigr)\mathcal{E}(1).
\label{eq:transfer}
\end{equation}
\end{lemma}
\begin{proof}
Immediate from Lemma~\ref{lem:point} and $\mathcal{E}$ decreasing
(Lemma~\ref{lem:bounds}).
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
Under \eqref{eq:conds}, $\mathcal{E}$ of \eqref{eq:point} is increasing in $a_\infty$
and, at $\rho=1$, maximised over $\kappa\ge0.2$ at $\kappa=0.2$; evaluating gives
$\mathcal{E}(1)=0.353$ at $\lambda=5$, $0.133$ at $\lambda=6$, $0.019$ at $\lambda=8$,
which is (a). Feeding $\mathcal{E}(\rho_i)$ into the first form of
\eqref{eq:transfer} --- which is $7$ to $10$ times smaller than the uniform second
form, since $\mathcal{E}$ decays like $e^{-\mu\rho}$ across the window --- gives (b)
and (c):
\begin{center}
\begin{tabular}{rrrrr}
\toprule
$\lambda\ge$ & $1/\lambda$ & $\mathcal{E}(1)$ & slope error & curvature error\\
\midrule
$4$ & $250$ ms & $0.96$  & $0.253$  & $0.446$\\
$5$ & $200$ ms & $0.353$ & $0.081$  & $0.145$\\
$6$ & $167$ ms & $0.133$ & $0.027$  & $0.049$\\
$8$ & $125$ ms & $0.019$ & $0.0030$ & $0.0058$\\
\bottomrule
\end{tabular}
\end{center}
\end{proof}

\begin{remark}
Lemma~\ref{lem:bounds} bounds the \emph{moments}; the observables are a slope and a
curvature, that is, \emph{derivatives}, and a bound on a function does not bound its
derivative. Lemma~\ref{lem:point} bridges the gap by bounding $\ln\nu$ itself,
pointwise, after which Lemma~\ref{lem:transfer} is elementary.
\end{remark}

\section{Predictions}\label{sec:pred}

We assume \eqref{eq:conds} throughout and work with $\nustat$, carrying
Theorem~\ref{thm:main}(b) as the error bar. The delay-period observations are:
\textbf{(L1)} the level rises with stimulus duration; \textbf{(L2)} the level rises
with motion coherence; \textbf{(L3)} the coherence effect on the level is
\emph{larger} after the long stimulus; \textbf{(S)} the log-rate slope is
$+0.867\,\mathrm{s}^{-1}$.

\begin{corollary}[L1, L2]\label{cor:P1}
\emph{From Proposition~\ref{prop:statrate}(iii), plus the saturation argument of the
main text, which concerns the distribution of $m$ and is untouched by
\eqref{eq:warm}.}
Duration and coherence each raise the fraction of trials whose maintained target has
reached the cap $\theta$, hence lower $d$ in the sense of first-order stochastic
dominance; since $\nustat$ is strictly decreasing in $d$, and averaging over trials
preserves the ordering, both raise the level. Predicted by \emph{both} models:
\emph{not discriminating}.
\end{corollary}

\begin{corollary}[L3]\label{cor:P2}
\emph{From Proposition~\ref{prop:statrate}(iii) and the same saturation argument.}
As the capped fraction tends to one the coherence levels converge, and duration raises
that fraction at every coherence, so the coherence effect on the level must
\emph{shrink} with duration. Predicted by both models and \emph{contradicted} by the
data: not discriminating, and a genuine disagreement.
\end{corollary}

\begin{corollary}[S]\label{cor:P3}
\emph{From Proposition~\ref{prop:statrate} and Theorem~\ref{thm:main}(b).}
Model~1 predicts a log-rate slope of exactly $0$
(Proposition~\ref{prop:statrate}(ii)), hence a measured slope of at most
$0.081\,\mathrm{s}^{-1}$ under \eqref{eq:conds}, and $0.027$ if $\lambda\ge6$.
Model~2 predicts $\beta a(\rho)/(\delta\sigma_H)>0$
(Proposition~\ref{prop:statrate}(i)), its only source of a ramp.
\emph{Discriminating: the observed $+0.867\,\mathrm{s}^{-1}$ exceeds Model~1's
ceiling by a factor of $10.8$, or $33$ if $\lambda\ge6$.}
\end{corollary}

\begin{remark}[what is no longer claimed]
Two delay-period statements that appear in the current draft are dropped here. The
\emph{shape} of the ramp no longer discriminates: under \eqref{eq:warm} there is no
variance filling to decay, both models predict a near-zero curvature (Model~1 exactly
$0$; Model~2 $-\frac12(\beta/\delta\sigma_H)^{2}\ge-0.034$ at the observed slope), and
the observed $-0.0405\pm0.0416$ is consistent with both. The dependence of the
\emph{slope on the conditions} is also dropped: it discriminates only at
$\lambda\ge6$, it was absent from the first version of the paper, and it is the only
statement requiring the unbinned point-process analysis. What remains is one
discriminating prediction, exactly stated, with an explicit error bar.
\end{remark}

\section{Consequences for the paper}

Adopting \eqref{eq:warm} removes from the appendix the convolution kernel $K$, the
integrals $\int K^{2}$ and $\int K'^{2}$, the auxiliary $W=K'-K^{3}/2\sigma_H^{2}$,
the log-concavity lemma, the non-stationary derivative of $\ln\nu$, and the transient
content of the present Corollaries A.2 and A.3, together with the point-process
analysis of the delay period. What remains is Proposition~\ref{prop:statrate},
Theorem~\ref{thm:main} with its conditions, and three corollaries; the proof of the
theorem can sit in a subsection or be deferred entirely.

The cost is that theory and simulation must agree, so the $1200$-draw parameter sweep,
the admissibility screen and every effect size would have to be recomputed under
\eqref{eq:warm}.
